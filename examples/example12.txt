# Initialize variables
x = 0;
y = 0;
diagonal = 0;
offDiagonal = 0;
max = 10;

# Nested loops with conditionals
while (x < max) {
    y = 0;
    while (y < max) {
        if (x == y) {
            diagonal = diagonal + 1;
        } else {
            offDiagonal = offDiagonal + 1;
        };
        y = y + 1;
    };
    x = x + 1;
}
# x = 10
# y = 10
# diagonal = 10
# offDiagonal = 90
# (checked in C++) * when variable isnt declared, its initialized as 0

# Conditional with loop inside
condition = 1;
if (condition == 1) {
    sum = 0;
    i = 1;
    while (i <= 5) {
        sum = sum + i;
        i = i + 1;
    }
} else {
    sum = 0;
}
# condition = 1
# sum = 15
# i = 6

# Loop with conditional inside
counter = 0;
total = 0;
while (counter < 10) {
    if (counter % 2 == 0) {
        total = total + counter;
    };
    counter = counter + 1;
}
# total = 20
# counter = 10
